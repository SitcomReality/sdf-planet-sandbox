// Fullscreen-triangle vertex shader. We render to a low-res offscreen target
// and let the browser bilinearly upscale to the canvas — a very cheap form of
// dynamic resolution scaling that's perfect for potatoes.
export const VERT_SRC = `#version 300 es
precision highp float;
out vec2 vUv;
void main() {
  // Single triangle covering the screen — no index buffer, no VBO.
  vec2 p = vec2((gl_VertexID == 1) ? 3.0 : -1.0,
                (gl_VertexID == 2) ? 3.0 : -1.0);
  vUv = p * 0.5 + 0.5;
  gl_Position = vec4(p, 0.0, 1.0);
}`;

// Fragment shader: sphere-traced SDF with aggressive optimizations.
//
// Per-pixel cost guards:
//   - MAX_STEPS capped at 40 (paper recommends 32-48 for mobile).
//   - DIST_THRESHOLD relaxed to 0.003 (paper recommends 0.002+).
//   - MIN_STEP early-out for grazing rays.
//   - Analytic ray-vs-world-sphere intersection to skip empty space before
//     marching (no wasted steps in the void around the planet).
//   - Edit loop bounded by uEditCount (CPU-side culled).
//
// Edits are read from a 1D-ish RGBA32F texture: 4 texels per edit.
//   texel 0: vec4(pos.xyz, kind)        // kind: 0=sphere 1=cube 2=torus 3=cyl
//   texel 1: vec4(size.xyz, op)         // op: +1 add, -1 subtract
//   texel 2: vec4(quat.xyzw)
//   texel 3: vec4(0)                    // reserved (e.g. material id later)
//
// The "ghost" preview is rendered as an extra additive edit with a flag to
// shade it cyan; we pack that in the reserved texel as ghost flag.
export const FRAG_SRC = `#version 300 es
precision highp float;

in vec2 vUv;
out vec4 fragColor;

uniform vec2  uResolution;
uniform vec3  uCamPos;
uniform mat3  uCamRot;        // camera orientation matrix (basis)
uniform float uFovScale;      // tan(fov/2)
uniform float uPlanetRadius;  // base sphere radius
uniform float uWorldRadius;   // bounding sphere for early-out (planetR + slack)
uniform sampler2D uEdits;     // edits texture
uniform int   uEditCount;
uniform int   uEditTexWidth;
uniform float uTime;
uniform vec3  uSunDir;

// Ghost preview (separate from edits texture so it never welds into geometry).
uniform int   uGhostKind;     // -1 = no ghost
uniform vec3  uGhostPos;
uniform vec3  uGhostSize;
uniform vec4  uGhostQuat;
uniform float uGhostOp;       // +1 add preview, -1 subtract preview

// --- quality knobs (would be uniforms in a settings menu) ---
const int   MAX_STEPS      = 40;
const float DIST_THRESHOLD = 0.003;
const float MIN_STEP       = 0.0008;
const float MAX_DIST       = 200.0;

// Smooth min / max (polynomial). Cheap and gives soft blending so player edits
// feel claylike rather than CSG-rigid.
float smin(float a, float b, float k) {
  float h = clamp(0.5 + 0.5 * (b - a) / k, 0.0, 1.0);
  return mix(b, a, h) - k * h * (1.0 - h);
}
float smax(float a, float b, float k) {
  float h = clamp(0.5 - 0.5 * (b - a) / k, 0.0, 1.0);
  return mix(b, a, h) + k * h * (1.0 - h);
}

// Rotate vector by inverse of quaternion (world->local).
vec3 qRotInv(vec4 q, vec3 v) {
  vec3 u = vec3(-q.x, -q.y, -q.z);
  return v + 2.0 * cross(u, cross(u, v) + q.w * v);
}

// SDF primitives (all centered at origin, in their local frame).
float sdSphere(vec3 p, float r) { return length(p) - r; }
float sdBox(vec3 p, vec3 b) {
  vec3 q = abs(p) - b;
  return length(max(q, 0.0)) + min(max(q.x, max(q.y, q.z)), 0.0);
}
float sdTorus(vec3 p, float R, float r) {
  vec2 q = vec2(length(p.xz) - R, p.y);
  return length(q) - r;
}
float sdCylinder(vec3 p, float r, float h) {
  vec2 d = abs(vec2(length(p.xz), p.y)) - vec2(r, h);
  return min(max(d.x, d.y), 0.0) + length(max(d, 0.0));
}

// Planet base: simple sphere. Terrain noise is added as cheap displacement
// only on this base SDF — cheaper than evaluating noise as separate edits per
// march step. (The "procedural terrain as edits" optimization is approximated
// by this single bumpy-sphere term; the same blending math applies.)
float terrainNoise(vec3 p) {
  // Two cheap sin-based octaves. ~6 muls + 3 sins. Way cheaper than fbm.
  vec3 n = normalize(p + 1e-5);
  float a = sin(n.x * 4.0 + n.y * 5.3) * cos(n.z * 4.7);
  float b = sin(n.x * 11.0) * sin(n.y * 9.7) * sin(n.z * 10.3);
  return a * 0.18 + b * 0.06;
}

float sdPlanet(vec3 p) {
  return length(p) - (uPlanetRadius + terrainNoise(p));
}

// Fetch one edit (4 consecutive texels) from the edits texture.
// Returns components via out params to avoid struct overhead.
void fetchEdit(int i, out vec4 a, out vec4 b, out vec4 c, out vec4 d) {
  int base = i * 4;
  int w = uEditTexWidth;
  ivec2 p0 = ivec2(base % w,       base / w);
  ivec2 p1 = ivec2((base+1) % w,  (base+1) / w);
  ivec2 p2 = ivec2((base+2) % w,  (base+2) / w);
  ivec2 p3 = ivec2((base+3) % w,  (base+3) / w);
  a = texelFetch(uEdits, p0, 0);
  b = texelFetch(uEdits, p1, 0);
  c = texelFetch(uEdits, p2, 0);
  d = texelFetch(uEdits, p3, 0);
}

// Planet base color (cheap latitude/slope tint).
vec3 planetColor(vec3 p, vec3 n) {
  float h = length(p) - uPlanetRadius;
  vec3 base = mix(vec3(0.42, 0.55, 0.25), vec3(0.55, 0.45, 0.30),
                  clamp(h * 4.0 + 0.5, 0.0, 1.0));
  float steep = 1.0 - clamp(dot(n, normalize(p)), 0.0, 1.0);
  return mix(base, vec3(0.45, 0.42, 0.40), smoothstep(0.4, 0.8, steep));
}

// Cheap normal-free planet color — used INSIDE mapScene so that planet/edit
// smin welds can blend smoothly in color too. Steep-slope rock tinting is
// applied later in main() (which has the normal) only when the pixel is
// purely planet (no edit influence) — this preserves the look while still
// giving us a usable RGB for the blend.
vec3 planetColorBase(vec3 p) {
  float h = length(p) - uPlanetRadius;
  return mix(vec3(0.42, 0.55, 0.25), vec3(0.55, 0.45, 0.30),
             clamp(h * 4.0 + 0.5, 0.0, 1.0));
}

// Full scene SDF: returns vec4(distance, rgb) plus a "planetW" weight via an
// out param (1=pure planet, 0=fully edit-dominated). Color blends in lock-step
// with the smin distance blend, so adjacent rainbow blobs fade smoothly into
// each other and into the planet color along the same isoline that smin
// smooths. Subtractions don't change color (host material wins inside carves).
// Ghost is handled separately in main() so it never welds into real geometry.
vec4 mapSceneFull(vec3 p, out float planetW) {
  float d = sdPlanet(p);
  vec3  col = planetColorBase(p);
  planetW = 1.0; // 1 = pure planet, 0 = pure edit
  const float K_ADD = 0.15;
  const float K_SUB = 0.20;
  for (int i = 0; i < 96; i++) {
    if (i >= uEditCount) break;
    vec4 a, b, c, dd;
    fetchEdit(i, a, b, c, dd);
    vec3 pos = a.xyz;
    int kind = int(a.w + 0.5);
    vec3 size = b.xyz;
    float op = b.w;
    vec4 q = c;
    vec3 editCol = dd.yzw;

    vec3 lp = qRotInv(q, p - pos);
    float ed;
    if (kind == 0)      ed = sdSphere(lp, size.x);
    else if (kind == 1) ed = sdBox(lp, size);
    else if (kind == 2) ed = sdTorus(lp, size.x, size.y);
    else                ed = sdCylinder(lp, size.x, size.y);

    if (op > 0.0) {
      // smin: h=1 -> all "a" (current d/col), h=0 -> all "b" (edit).
      float k = K_ADD;
      float h = clamp(0.5 + 0.5 * (ed - d) / k, 0.0, 1.0);
      d   = mix(ed, d, h) - k * h * (1.0 - h);
      col = mix(editCol, col, h);
      planetW *= h; // every edit reduces "planet purity" the more it dominates
    } else {
      // Smooth subtract: smax(d, -ed, K_SUB). Color unchanged — carving doesn't
      // recolor the host material.
      // NOTE: smax(a,b,k) = mix(b, a, h) + k*h*(1-h)  with h = clamp(0.5+0.5*(a-b)/k,0,1)
      // The mix args MUST be (b, a) not (a, b) — otherwise points far outside
      // the carved region read as deep-inside-surface and the entire world
      // collapses into a smooth bounding sphere.
      float k = K_SUB;
      float a2 = d, b2 = -ed;
      float h2 = clamp(0.5 + 0.5 * (a2 - b2) / k, 0.0, 1.0);
      d = mix(b2, a2, h2) + k * h2 * (1.0 - h2);
    }
  }
  return vec4(d, col);
}

// Convenience wrapper used by the marcher (planet-weight not needed there).
vec4 mapScene(vec3 p) {
  float w;
  return mapSceneFull(p, w);
}

// Ghost SDF — only the ghost edit, no scene composition. Returns distance.
float mapGhost(vec3 p) {
  if (uGhostKind < 0) return 1e9;
  vec3 lp = qRotInv(uGhostQuat, p - uGhostPos);
  if (uGhostKind == 0)      return sdSphere(lp, uGhostSize.x);
  else if (uGhostKind == 1) return sdBox(lp, uGhostSize);
  else if (uGhostKind == 2) return sdTorus(lp, uGhostSize.x, uGhostSize.y);
  else                      return sdCylinder(lp, uGhostSize.x, uGhostSize.y);
}

// Combined SDF that the marcher actually steps along: scene UNIONed with ghost.
// Returning a float keeps the inner loop hot and lets the compiler keep
// everything in registers. We resolve "which surface won" once at the end.
float mapAll(vec3 p) {
  float ds = mapScene(p).x;
  float dg = mapGhost(p);
  return min(ds, dg);
}

// Cheap normal via tetrahedral offsets (4 evals instead of 6).
vec3 calcNormal(vec3 p) {
  const float h = 0.01;
  const vec2 k = vec2(1.0, -1.0);
  return normalize(
    k.xyy * mapAll(p + k.xyy * h) +
    k.yyx * mapAll(p + k.yyx * h) +
    k.yxy * mapAll(p + k.yxy * h) +
    k.xxx * mapAll(p + k.xxx * h));
}

// Ray vs sphere — analytic, gives near/far entry distances. Used to skip the
// empty void before the planet (huge perf win when looking outward).
vec2 raySphere(vec3 ro, vec3 rd, float R) {
  float b = dot(ro, rd);
  float c = dot(ro, ro) - R * R;
  float h = b * b - c;
  if (h < 0.0) return vec2(-1.0);
  h = sqrt(h);
  return vec2(-b - h, -b + h);
}

void main() {
  // Build view ray.
  vec2 uv = (vUv * 2.0 - 1.0);
  uv.x *= uResolution.x / uResolution.y;
  vec3 rd = normalize(uCamRot * vec3(uv * uFovScale, -1.0));
  vec3 ro = uCamPos;

  // World-bounding-sphere early-out: if the ray misses, render sky immediately.
  vec2 hit = raySphere(ro, rd, uWorldRadius);
  if (hit.y < 0.0) {
    // Sky: cheap 2-color gradient based on ray.y.
    vec3 sky = mix(vec3(0.55, 0.72, 0.95), vec3(0.18, 0.32, 0.62),
                   clamp(rd.y * 0.5 + 0.5, 0.0, 1.0));
    // Tiny sun disc.
    float sun = smoothstep(0.997, 0.9995, dot(rd, uSunDir));
    sky += vec3(1.0, 0.95, 0.85) * sun;
    fragColor = vec4(sky, 1.0);
    return;
  }

  // Start march at sphere entry (or current position if inside). Also clamp the
  // start to "just behind the camera" so the ghost remains visible even when
  // the player pulls it very close (otherwise the ghost would clip the start
  // sphere and disappear).
  float t = max(hit.x, 0.0);
  float tMax = hit.y + 1.0;
  // If the ghost might be in front of the world bounding sphere entry, start
  // at 0 to ensure we don't skip past it.
  if (uGhostKind >= 0) t = 0.0;

  vec3 p;
  bool hitSurface = false;

  for (int i = 0; i < MAX_STEPS; i++) {
    p = ro + rd * t;
    float d = mapAll(p);
    if (d < DIST_THRESHOLD) { hitSurface = true; break; }
    float step = max(d, MIN_STEP);
    t += step;
    if (t > tMax) break;
  }

  if (!hitSurface) {
    vec3 sky = mix(vec3(0.55, 0.72, 0.95), vec3(0.18, 0.32, 0.62),
                   clamp(rd.y * 0.5 + 0.5, 0.0, 1.0));
    float sun = smoothstep(0.997, 0.9995, dot(rd, uSunDir));
    sky += vec3(1.0, 0.95, 0.85) * sun;
    fragColor = vec4(sky, 1.0);
    return;
  }

  vec3 n = calcNormal(p);

  // Decide which surface this pixel belongs to: scene or ghost.
  float planetW;
  vec4 sceneRes = mapSceneFull(p, planetW);
  float dGhost  = mapGhost(p);
  bool isGhost  = dGhost < sceneRes.x;

  vec3 base;
  if (isGhost) {
    // Ghost preview — pulsing tint. Color hints at op: cyan = add, red = sub.
    float pulse = 0.5 + 0.5 * sin(uTime * 4.0);
    vec3 addCol = mix(vec3(0.30, 0.85, 1.00), vec3(0.65, 1.00, 1.00), pulse);
    vec3 subCol = mix(vec3(1.00, 0.35, 0.35), vec3(1.00, 0.65, 0.55), pulse);
    base = (uGhostOp > 0.0) ? addCol : subCol;
  } else {
    // Use smin-blended color from mapSceneFull (rainbow fades cleanly into
    // adjacent blobs and into the planet base color). Then layer slope-rock
    // tint, weighted by planet purity so rocks only show where the planet
    // dominates — edit-dominated regions keep their saturated color.
    base = sceneRes.yzw;
    float steep = 1.0 - clamp(dot(n, normalize(p)), 0.0, 1.0);
    float rockMix = smoothstep(0.4, 0.8, steep) * planetW;
    base = mix(base, vec3(0.45, 0.42, 0.40), rockMix);
  }

  // Lighting: 1 directional + cheap ambient + faux AO from march iteration count.
  float diff = max(dot(n, uSunDir), 0.0);
  float amb  = 0.35 + 0.25 * (n.y * 0.5 + 0.5);
  vec3 col = base * (amb + diff * 0.7);

  // Distance fog (fades into sky color).
  float fog = 1.0 - exp(-t * 0.012);
  vec3 sky = mix(vec3(0.55, 0.72, 0.95), vec3(0.18, 0.32, 0.62),
                 clamp(rd.y * 0.5 + 0.5, 0.0, 1.0));
  col = mix(col, sky, fog);

  fragColor = vec4(col, 1.0);
}`;
