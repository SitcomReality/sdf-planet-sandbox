import { useEffect, useRef, useState, useCallback } from "react";
import { SdfEngine } from "./sdf/engine";
import type { ShapeKind } from "./sdf/bvh";

const SHAPES: { kind: ShapeKind; name: string; key: string; icon: string }[] = [
  { kind: 0, name: "Sphere",   key: "1", icon: "●" },
  { kind: 1, name: "Cube",     key: "2", icon: "■" },
  { kind: 2, name: "Torus",    key: "3", icon: "◯" },
  { kind: 3, name: "Cylinder", key: "4", icon: "▮" },
];

export default function App() {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const engineRef = useRef<SdfEngine | null>(null);
  const [loadingLabel, setLoadingLabel] = useState<string>("Booting…");
  const [loadingFrac, setLoadingFrac] = useState<number>(0);
  const [ready, setReady] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [selected, setSelected] = useState<ShapeKind>(0);
  const [pointerLocked, setPointerLocked] = useState(false);
  const [stats, setStats] = useState({ totalEdits: 0, fps: 0 });
  const [renderScale, setRenderScale] = useState(0.5);

  // Persistent input state via refs (no re-renders per frame).
  const keys = useRef<Record<string, boolean>>({});
  const eHeld = useRef(false);

  // Init engine.
  useEffect(() => {
    if (!canvasRef.current) return;
    let engine: SdfEngine;
    try {
      engine = new SdfEngine(canvasRef.current, {
        onLoadingProgress: (label, frac) => {
          setLoadingLabel(label);
          setLoadingFrac(frac);
        },
        onError: (msg) => setError(msg),
      });
    } catch (e) {
      return;
    }
    engineRef.current = engine;

    // Defer "ready" so loading bar reaches 100% visually.
    const t = setTimeout(() => setReady(true), 350);
    return () => { clearTimeout(t); };
  }, []);

  // Render loop (only redraws on dirty).
  useEffect(() => {
    if (!ready || !engineRef.current) return;
    let raf = 0;
    let lastT = performance.now();
    let frameCount = 0;
    let fpsAcc = 0;
    let fpsT = lastT;

    const tick = (now: number) => {
      const engine = engineRef.current;
      if (!engine) return;
      const dt = Math.min(0.05, (now - lastT) / 1000);
      lastT = now;

      // Movement.
      const f = (keys.current["w"] ? 1 : 0) - (keys.current["s"] ? 1 : 0);
      const s = (keys.current["d"] ? 1 : 0) - (keys.current["a"] ? 1 : 0);
      const u = (keys.current[" "] ? 1 : 0) - (keys.current["shift"] ? 1 : 0);
      if (f || s || u) engine.move(f, s, u, dt);

      // (Ghost is permanently visible; no per-frame raycast needed — the
      // engine recomputes its position from camera state when it renders.)

      const drew = engine.renderIfDirty(now);
      if (drew) fpsAcc++;
      frameCount++;
      if (now - fpsT > 500) {
        const fps = (fpsAcc * 1000) / (now - fpsT);
        setStats((p) => ({ ...p, fps: Math.round(fps), totalEdits: engine.edits.length }));
        fpsAcc = 0; fpsT = now;
      }
      raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [ready]);

  // Keyboard.
  useEffect(() => {
    const down = (e: KeyboardEvent) => {
      const k = e.key.toLowerCase();
      keys.current[k] = true;
      if (k === "e") eHeld.current = true;
      if (k >= "1" && k <= "4") {
        const kind = (parseInt(k, 10) - 1) as ShapeKind;
        engineRef.current?.setShape(kind);
        setSelected(kind);
      }
      // Prevent space scrolling.
      if (k === " ") e.preventDefault();
    };
    const up = (e: KeyboardEvent) => {
      const k = e.key.toLowerCase();
      keys.current[k] = false;
      if (k === "e") eHeld.current = false;
    };
    window.addEventListener("keydown", down);
    window.addEventListener("keyup", up);
    return () => {
      window.removeEventListener("keydown", down);
      window.removeEventListener("keyup", up);
    };
  }, []);

  // Mouse.
  const handleCanvasClick = useCallback(() => {
    if (!pointerLocked && canvasRef.current) {
      canvasRef.current.requestPointerLock();
    }
  }, [pointerLocked]);

  useEffect(() => {
    const onLockChange = () => {
      setPointerLocked(document.pointerLockElement === canvasRef.current);
    };
    document.addEventListener("pointerlockchange", onLockChange);
    return () => document.removeEventListener("pointerlockchange", onLockChange);
  }, []);

  useEffect(() => {
    if (!pointerLocked) return;
    const onMove = (e: MouseEvent) => {
      const eng = engineRef.current;
      if (!eng) return;
      if (eHeld.current) {
        eng.rotateGhost(e.movementX, e.movementY);
      } else {
        eng.look(e.movementX, e.movementY);
      }
    };
    const onMouseDown = (e: MouseEvent) => {
      const eng = engineRef.current;
      if (!eng) return;
      if (e.button === 0) eng.performAction(false);
      else if (e.button === 2) {
        // Show red preview while RMB is held, then commit on release.
        eng.setSubtractMode(true);
      }
    };
    const onMouseUp = (e: MouseEvent) => {
      const eng = engineRef.current;
      if (!eng) return;
      if (e.button === 2) {
        eng.performAction(true);
        eng.setSubtractMode(false);
      }
    };
    const onWheel = (e: WheelEvent) => {
      const eng = engineRef.current;
      if (!eng) return;
      e.preventDefault();
      // Negative deltaY = scroll up = push ghost FARTHER (intuitive).
      eng.adjustGhostDistance(-e.deltaY * 0.003);
    };
    const onContext = (e: Event) => e.preventDefault();
    document.addEventListener("mousemove", onMove);
    document.addEventListener("mousedown", onMouseDown);
    document.addEventListener("mouseup", onMouseUp);
    document.addEventListener("wheel", onWheel, { passive: false });
    document.addEventListener("contextmenu", onContext);
    return () => {
      document.removeEventListener("mousemove", onMove);
      document.removeEventListener("mousedown", onMouseDown);
      document.removeEventListener("mouseup", onMouseUp);
      document.removeEventListener("wheel", onWheel);
      document.removeEventListener("contextmenu", onContext);
    };
  }, [pointerLocked]);

  const onChangeQuality = (s: number) => {
    setRenderScale(s);
    engineRef.current?.setRenderScale(s);
  };

  return (
    <div className="w-screen h-screen relative overflow-hidden bg-slate-900 text-white select-none font-sans">
      <canvas
        ref={canvasRef}
        onClick={handleCanvasClick}
        className="absolute inset-0 w-full h-full block cursor-crosshair"
      />

      {error && (
        <div className="absolute inset-0 flex items-center justify-center bg-black/80 z-50 p-8">
          <div className="max-w-md text-center">
            <div className="text-2xl mb-2 text-red-400">⚠️ Could not start</div>
            <div className="text-sm opacity-80 whitespace-pre-wrap">{error}</div>
          </div>
        </div>
      )}

      {!ready && !error && (
        <div className="absolute inset-0 flex flex-col items-center justify-center bg-gradient-to-b from-slate-900 to-slate-800 z-40">
          <div className="text-3xl font-bold mb-2 tracking-tight">SDF Planet</div>
          <div className="text-sm opacity-70 mb-8">Tiny world sandbox</div>
          <div className="w-72 h-2 bg-slate-700 rounded-full overflow-hidden mb-3">
            <div
              className="h-full bg-emerald-400 transition-all duration-200"
              style={{ width: `${Math.round(loadingFrac * 100)}%` }}
            />
          </div>
          <div className="text-xs opacity-60">{loadingLabel}</div>
          <div className="text-xs opacity-40 mt-6 max-w-sm text-center px-6">
            Pre-evaluating distance field, allocating brick atlas, and seeding terrain noise…
          </div>
        </div>
      )}

      {ready && !pointerLocked && (
        <div className="absolute inset-0 flex items-center justify-center bg-black/60 z-30 pointer-events-none">
          <div className="text-center pointer-events-none">
            <div className="text-2xl mb-3 font-semibold">Click to play</div>
            <div className="text-sm opacity-80 leading-relaxed">
              <kbd className="px-2 py-0.5 bg-white/15 rounded">WASD</kbd> move &nbsp;
              <kbd className="px-2 py-0.5 bg-white/15 rounded">Space</kbd> up &nbsp;
              <kbd className="px-2 py-0.5 bg-white/15 rounded">Shift</kbd> down &nbsp;
              <kbd className="px-2 py-0.5 bg-white/15 rounded">Mouse</kbd> look<br/>
              <span className="block mt-2">
                <kbd className="px-2 py-0.5 bg-white/15 rounded">Left&nbsp;Click</kbd> add &nbsp;
                <kbd className="px-2 py-0.5 bg-white/15 rounded">Right&nbsp;Click</kbd> subtract
              </span>
              <span className="block mt-2">
                <kbd className="px-2 py-0.5 bg-white/15 rounded">Scroll</kbd> push/pull ghost · hold <kbd className="px-2 py-0.5 bg-white/15 rounded">E</kbd> + mouse to rotate it
              </span>
              <span className="block mt-2">
                <kbd className="px-2 py-0.5 bg-white/15 rounded">1-4</kbd> select shape · <kbd className="px-2 py-0.5 bg-white/15 rounded">Esc</kbd> release mouse
              </span>
            </div>
          </div>
        </div>
      )}

      {/* Crosshair */}
      {ready && pointerLocked && (
        <div className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 pointer-events-none">
          <div className="w-3 h-3 border border-white/80 rotate-45" />
        </div>
      )}

      {/* Top-left HUD */}
      {ready && (
        <div className="absolute top-3 left-3 text-xs bg-black/40 backdrop-blur-sm rounded px-3 py-2 space-y-0.5 pointer-events-auto">
          <div>FPS: <span className="font-mono">{stats.fps}</span></div>
          <div>Edits: <span className="font-mono">{stats.totalEdits}</span></div>
          <div className="flex items-center gap-2 mt-1">
            <span>Quality:</span>
            <select
              value={renderScale}
              onChange={(e) => onChangeQuality(parseFloat(e.target.value))}
              className="bg-slate-800 border border-white/20 rounded px-1 py-0.5 text-xs"
            >
              <option value={0.35}>Potato (35%)</option>
              <option value={0.5}>Low (50%)</option>
              <option value={0.75}>Medium (75%)</option>
              <option value={1.0}>High (100%)</option>
            </select>
          </div>
        </div>
      )}

      {/* Action bar */}
      {ready && (
        <div className="absolute bottom-6 left-1/2 -translate-x-1/2 flex gap-2 bg-black/45 backdrop-blur-sm rounded-xl p-2 border border-white/10">
          {SHAPES.map((s) => (
            <button
              key={s.kind}
              onClick={(e) => {
                e.stopPropagation();
                engineRef.current?.setShape(s.kind);
                setSelected(s.kind);
              }}
              className={
                "w-16 h-16 rounded-lg flex flex-col items-center justify-center transition-all border-2 " +
                (selected === s.kind
                  ? "bg-emerald-500/30 border-emerald-400 shadow-lg shadow-emerald-500/30"
                  : "bg-white/5 border-white/10 hover:bg-white/10")
              }
              title={`${s.name} (${s.key})`}
            >
              <div className="text-2xl leading-none">{s.icon}</div>
              <div className="text-[10px] opacity-80 mt-1">{s.name}</div>
              <div className="text-[9px] opacity-50">[{s.key}]</div>
            </button>
          ))}
        </div>
      )}

      {/* Bottom-right hint */}
      {ready && pointerLocked && (
        <div className="absolute bottom-6 right-6 text-xs opacity-70 text-right leading-relaxed pointer-events-none">
          <div>L-Click: <span className="text-emerald-300">add</span> · R-Click: <span className="text-rose-300">subtract</span> · Scroll: distance</div>
          <div>Hold <kbd className="px-1.5 py-0.5 bg-white/15 rounded">E</kbd> + mouse to rotate ghost</div>
        </div>
      )}
    </div>
  );
}
