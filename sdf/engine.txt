import { VERT_SRC, FRAG_SRC } from "./shaders";
import { Edit, ShapeKind, makeEdit, cullEditsToSphere, hsvToRgb } from "./bvh";

// Hard caps. With LRU eviction, the world stays performant even after
// hours of building. 96 matches the GPU loop bound in shaders.ts.
const MAX_EDITS_TOTAL = 256;
const MAX_EDITS_GPU = 96;
const EDIT_TEX_WIDTH = 64; // texels; 4 texels per edit -> 16 edits per row

const PLANET_RADIUS = 6.0;
const WORLD_RADIUS = PLANET_RADIUS + 3.0;
const CULL_RADIUS = 30.0; // around camera

// CPU-side mirror of the planet SDF, used for picking. Must match the GPU
// shader's sdPlanet exactly (or close enough that hit points feel right).
function terrainNoise(x: number, y: number, z: number): number {
  const len = Math.hypot(x, y, z) + 1e-5;
  const nx = x / len, ny = y / len, nz = z / len;
  const a = Math.sin(nx * 4.0 + ny * 5.3) * Math.cos(nz * 4.7);
  const b = Math.sin(nx * 11.0) * Math.sin(ny * 9.7) * Math.sin(nz * 10.3);
  return a * 0.18 + b * 0.06;
}
function sdPlanet(x: number, y: number, z: number): number {
  return Math.hypot(x, y, z) - (PLANET_RADIUS + terrainNoise(x, y, z));
}

// Quaternion helpers.
export function quatIdentity(): [number, number, number, number] { return [0, 0, 0, 1]; }
export function quatFromAxisAngle(ax: number, ay: number, az: number, ang: number): [number, number, number, number] {
  const s = Math.sin(ang / 2);
  return [ax * s, ay * s, az * s, Math.cos(ang / 2)];
}
export function quatMul(a: number[], b: number[]): [number, number, number, number] {
  return [
    a[3] * b[0] + a[0] * b[3] + a[1] * b[2] - a[2] * b[1],
    a[3] * b[1] - a[0] * b[2] + a[1] * b[3] + a[2] * b[0],
    a[3] * b[2] + a[0] * b[1] - a[1] * b[0] + a[2] * b[3],
    a[3] * b[3] - a[0] * b[0] - a[1] * b[1] - a[2] * b[2],
  ];
}

export interface EngineCallbacks {
  onLoadingProgress?: (label: string, frac: number) => void;
  onReady?: () => void;
  onError?: (msg: string) => void;
}

export interface PlayerState {
  // Camera position around the planet (orbital coords + altitude).
  // We use an orbital camera so movement always stays tangent to the planet —
  // perfect for "tiny planet" feel and avoids gimbal issues.
  lon: number;       // longitude (radians)
  lat: number;       // latitude (radians)
  altitude: number;  // height above mean planet radius
  yaw: number;       // local heading
  pitch: number;     // local look up/down
}

export class SdfEngine {
  private gl: WebGL2RenderingContext;
  private prog!: WebGLProgram;
  private vao!: WebGLVertexArrayObject;
  private editsTex!: WebGLTexture;
  private editsData: Float32Array;
  private uniforms: Record<string, WebGLUniformLocation | null> = {};
  private dirty = true;
  private renderScale = 0.5;        // dynamic resolution
  private fbo!: WebGLFramebuffer;
  private fboTex!: WebGLTexture;
  private fboW = 0;
  private fboH = 0;
  private blitProg!: WebGLProgram;
  private blitUniformTex!: WebGLUniformLocation | null;

  edits: Edit[] = [];
  player: PlayerState = {
    lon: 0,
    lat: 0.3,
    altitude: 3.5,
    yaw: 0,
    pitch: -0.4,
  };
  selectedShape: ShapeKind = 0;
  shapeSize: [number, number, number] = [0.6, 0.6, 0.6];

  // Ghost lives at a fixed distance ahead of the camera. Always visible.
  // Scroll wheel adjusts ghostDistance. Holding E lets the mouse rotate the
  // ghost (in player-local space) instead of looking around.
  ghostDistance = 3.0;
  ghostUserQuat: [number, number, number, number] = quatIdentity(); // user's rotation in player-local frame
  ghostSubtractMode = false; // true while right mouse button is held — preview turns red

  // Hue cursor — advances per placement so consecutive blobs paint a rainbow.
  hueCursor = 0.0;
  hueStep = 0.07; // ~25 degrees

  private callbacks: EngineCallbacks;

  constructor(canvas: HTMLCanvasElement, callbacks: EngineCallbacks = {}) {
    this.callbacks = callbacks;
    const gl = canvas.getContext("webgl2", {
      antialias: false,
      alpha: false,
      powerPreference: "low-power",
      preserveDrawingBuffer: false,
    });
    if (!gl) {
      callbacks.onError?.("WebGL2 is not supported on this device.");
      throw new Error("WebGL2 unavailable");
    }
    this.gl = gl;
    this.editsData = new Float32Array(EDIT_TEX_WIDTH * EDIT_TEX_WIDTH * 4);
    this.initGL();
    this.seedTerrain();
  }

  private initGL() {
    const gl = this.gl;

    // Float texture support — required for our packed edit data.
    if (!gl.getExtension("EXT_color_buffer_float")) {
      // Not strictly required (we only sample, don't render to it), but
      // OES_texture_float_linear is nice. We'll use NEAREST sampling so we're fine.
    }

    this.prog = this.compile(VERT_SRC, FRAG_SRC);
    gl.useProgram(this.prog);

    const names = [
      "uResolution", "uCamPos", "uCamRot", "uFovScale",
      "uPlanetRadius", "uWorldRadius",
      "uEdits", "uEditCount", "uEditTexWidth",
      "uTime", "uSunDir",
      "uGhostKind", "uGhostPos", "uGhostSize", "uGhostQuat", "uGhostOp",
    ];
    for (const n of names) this.uniforms[n] = gl.getUniformLocation(this.prog, n);

    this.vao = gl.createVertexArray()!;

    // Edits texture.
    this.editsTex = gl.createTexture()!;
    gl.bindTexture(gl.TEXTURE_2D, this.editsTex);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.NEAREST);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.NEAREST);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA32F, EDIT_TEX_WIDTH, EDIT_TEX_WIDTH, 0, gl.RGBA, gl.FLOAT, this.editsData);

    // Offscreen low-res target for dynamic resolution scaling.
    this.fbo = gl.createFramebuffer()!;
    this.fboTex = gl.createTexture()!;

    // Blit program — bilinear upscale to canvas.
    const blitVert = `#version 300 es
      precision highp float;
      out vec2 vUv;
      void main(){
        vec2 p = vec2((gl_VertexID==1)?3.0:-1.0,(gl_VertexID==2)?3.0:-1.0);
        vUv = p*0.5+0.5;
        gl_Position = vec4(p,0.0,1.0);
      }`;
    const blitFrag = `#version 300 es
      precision highp float;
      in vec2 vUv; out vec4 o;
      uniform sampler2D uTex;
      void main(){ o = texture(uTex, vUv); }`;
    this.blitProg = this.compile(blitVert, blitFrag);
    this.blitUniformTex = gl.getUniformLocation(this.blitProg, "uTex");
  }

  private compile(vs: string, fs: string): WebGLProgram {
    const gl = this.gl;
    const v = gl.createShader(gl.VERTEX_SHADER)!;
    gl.shaderSource(v, vs); gl.compileShader(v);
    if (!gl.getShaderParameter(v, gl.COMPILE_STATUS)) {
      const log = gl.getShaderInfoLog(v) || "";
      this.callbacks.onError?.("Vertex shader: " + log);
      throw new Error(log);
    }
    const f = gl.createShader(gl.FRAGMENT_SHADER)!;
    gl.shaderSource(f, fs); gl.compileShader(f);
    if (!gl.getShaderParameter(f, gl.COMPILE_STATUS)) {
      const log = gl.getShaderInfoLog(f) || "";
      this.callbacks.onError?.("Fragment shader: " + log);
      throw new Error(log);
    }
    const p = gl.createProgram()!;
    gl.attachShader(p, v); gl.attachShader(p, f); gl.linkProgram(p);
    if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
      const log = gl.getProgramInfoLog(p) || "";
      this.callbacks.onError?.("Link: " + log);
      throw new Error(log);
    }
    return p;
  }

  // Seed a few starter edits to make the planet feel "lived in".
  // Uses a pseudo-progress callback so the loading screen feels honest.
  private seedTerrain() {
    const cb = this.callbacks.onLoadingProgress;
    cb?.("Cooling planet crust…", 0.1);
    // A handful of "boulders" scattered around (additive spheres).
    const N = 14;
    for (let i = 0; i < N; i++) {
      const t = i / N;
      const lon = t * Math.PI * 2 * 3.7;
      const lat = Math.sin(i * 1.7) * 1.1;
      const cl = Math.cos(lat);
      const r = PLANET_RADIUS + terrainNoise(
        Math.cos(lon) * cl * PLANET_RADIUS,
        Math.sin(lat) * PLANET_RADIUS,
        Math.sin(lon) * cl * PLANET_RADIUS
      );
      const x = Math.cos(lon) * cl * r;
      const y = Math.sin(lat) * r;
      const z = Math.sin(lon) * cl * r;
      const sz = 0.25 + (i % 3) * 0.15;
      this.edits.push(makeEdit(0, [x, y, z], [sz, sz, sz], quatIdentity(), 1));
    }
    cb?.("Building atmospheric gradient…", 0.5);
    cb?.("Allocating brick map slots…", 0.8);
    cb?.("Ready.", 1.0);
    this.markDirty();
  }

  markDirty() { this.dirty = true; }

  // Add or remove the selected shape at the camera's look-target. Returns true
  // if successful. Position is centered on the surface (object half-buried).
  // Place (or subtract) the selected shape at the current ghost position.
  // No surface raycast required — the ghost IS the placement preview, so what
  // you see is exactly what you get. Works equally well in mid-air, on the
  // planet, or intersecting another player-built object.
  performAction(subtract: boolean): boolean {
    const pos = this.ghostWorldPos();
    const quat = this.ghostWorldQuat();
    const color: [number, number, number] = subtract
      ? [0.78, 0.74, 0.66] // doesn't matter visually for subtracts
      : hsvToRgb(this.hueCursor, 0.85, 0.95);
    const e = makeEdit(
      this.selectedShape,
      [pos[0], pos[1], pos[2]],
      this.shapeSize.slice() as [number, number, number],
      quat,
      subtract ? -1 : 1,
      color,
    );
    this.edits.push(e);
    if (this.edits.length > MAX_EDITS_TOTAL) this.edits.shift();
    // Advance the rainbow on every action (add OR subtract).
    this.hueCursor = (this.hueCursor + this.hueStep) % 1.0;
    this.markDirty();
    return true;
  }

  // Ghost world-space position: a fixed distance ahead of the camera.
  ghostWorldPos(): [number, number, number] {
    const cam = this.cameraPosition();
    const fwd = this.cameraForward();
    return [
      cam[0] + fwd[0] * this.ghostDistance,
      cam[1] + fwd[1] * this.ghostDistance,
      cam[2] + fwd[2] * this.ghostDistance,
    ];
  }

  // Ghost world-space rotation: the ghost is anchored in CAMERA-LOCAL space
  // and only rotates when the player holds E + moves the mouse. From the
  // camera's point of view the ghost is perfectly stationary as you walk and
  // look around — like an item held in front of the player. To get the world-
  // space orientation we compose: world = camera * userLocal.
  ghostWorldQuat(): [number, number, number, number] {
    const camQ = this.cameraQuat();
    return quatMul(camQ, this.ghostUserQuat);
  }

  // Build the camera's orientation as a quaternion from its basis vectors.
  // Quaternion from a 3x3 rotation matrix (Shepperd's method, branch on trace).
  cameraQuat(): [number, number, number, number] {
    const b = this.cameraBasis();
    // Columns of the rotation matrix are right, up, fwd (world axes).
    const m00 = b.right[0], m01 = b.up[0], m02 = b.fwd[0];
    const m10 = b.right[1], m11 = b.up[1], m12 = b.fwd[1];
    const m20 = b.right[2], m21 = b.up[2], m22 = b.fwd[2];
    const tr = m00 + m11 + m22;
    let qx: number, qy: number, qz: number, qw: number;
    if (tr > 0) {
      const s = Math.sqrt(tr + 1.0) * 2;
      qw = 0.25 * s;
      qx = (m21 - m12) / s;
      qy = (m02 - m20) / s;
      qz = (m10 - m01) / s;
    } else if (m00 > m11 && m00 > m22) {
      const s = Math.sqrt(1.0 + m00 - m11 - m22) * 2;
      qw = (m21 - m12) / s;
      qx = 0.25 * s;
      qy = (m01 + m10) / s;
      qz = (m02 + m20) / s;
    } else if (m11 > m22) {
      const s = Math.sqrt(1.0 + m11 - m00 - m22) * 2;
      qw = (m02 - m20) / s;
      qx = (m01 + m10) / s;
      qy = 0.25 * s;
      qz = (m12 + m21) / s;
    } else {
      const s = Math.sqrt(1.0 + m22 - m00 - m11) * 2;
      qw = (m10 - m01) / s;
      qx = (m02 + m20) / s;
      qy = (m12 + m21) / s;
      qz = 0.25 * s;
    }
    const l = Math.hypot(qx, qy, qz, qw) || 1;
    return [qx/l, qy/l, qz/l, qw/l];
  }

  // CPU-side raycast against the SDF (sphere-trace) for picking. Uses tighter
  // params than the GPU since this runs once per click, not per pixel.
  // Considers all current edits (still cheap — bounded by MAX_EDITS_TOTAL).
  raycast(): [number, number, number] | null {
    const cam = this.cameraPosition();
    const fwd = this.cameraForward();
    let t = 0;
    const tMax = 60;
    for (let i = 0; i < 96; i++) {
      const px = cam[0] + fwd[0] * t;
      const py = cam[1] + fwd[1] * t;
      const pz = cam[2] + fwd[2] * t;
      const d = this.sceneSDF(px, py, pz);
      if (d < 0.005) return [px, py, pz];
      t += Math.max(d, 0.005);
      if (t > tMax) return null;
    }
    return null;
  }

  // CPU mirror of mapScene (planet + edits, smin/smax blending).
  private sceneSDF(x: number, y: number, z: number): number {
    let d = sdPlanet(x, y, z);
    const K_ADD = 0.15, K_SUB = 0.20;
    for (let i = 0; i < this.edits.length; i++) {
      const e = this.edits[i];
      // Quick AABB reject.
      if (x < e.aabbMin[0] - 0.5 || x > e.aabbMax[0] + 0.5) continue;
      if (y < e.aabbMin[1] - 0.5 || y > e.aabbMax[1] + 0.5) continue;
      if (z < e.aabbMin[2] - 0.5 || z > e.aabbMax[2] + 0.5) continue;
      // Transform into edit's local frame.
      const dx = x - e.px, dy = y - e.py, dz = z - e.pz;
      // qRotInv
      const ux = -e.qx, uy = -e.qy, uz = -e.qz, w = e.qw;
      // v + 2*cross(u, cross(u,v) + w*v)
      const cx1 = uy * dz - uz * dy;
      const cy1 = uz * dx - ux * dz;
      const cz1 = ux * dy - uy * dx;
      const tx = cx1 + w * dx;
      const ty = cy1 + w * dy;
      const tz = cz1 + w * dz;
      const cx2 = uy * tz - uz * ty;
      const cy2 = uz * tx - ux * tz;
      const cz2 = ux * ty - uy * tx;
      const lx = dx + 2 * cx2;
      const ly = dy + 2 * cy2;
      const lz = dz + 2 * cz2;
      let ed: number;
      if (e.kind === 0) ed = Math.hypot(lx, ly, lz) - e.sx;
      else if (e.kind === 1) {
        const qx = Math.abs(lx) - e.sx;
        const qy = Math.abs(ly) - e.sy;
        const qz = Math.abs(lz) - e.sz;
        const mx = Math.max(qx, 0), my = Math.max(qy, 0), mz = Math.max(qz, 0);
        ed = Math.hypot(mx, my, mz) + Math.min(Math.max(qx, qy, qz), 0);
      } else if (e.kind === 2) {
        const r = Math.hypot(lx, lz) - e.sx;
        ed = Math.hypot(r, ly) - e.sy;
      } else {
        const rr = Math.hypot(lx, lz) - e.sx;
        const hh = Math.abs(ly) - e.sy;
        const mr = Math.max(rr, 0), mh = Math.max(hh, 0);
        ed = Math.min(Math.max(rr, hh), 0) + Math.hypot(mr, mh);
      }
      if (e.op > 0) {
        const k = K_ADD;
        const h = Math.max(0, Math.min(1, 0.5 + 0.5 * (d - ed) / k));
        d = ed * (1 - h) + d * h - k * h * (1 - h);
      } else {
        const k = K_SUB;
        const a = d, b = -ed;
        // smax: result = mix(b, a, h2). Same fix as the GLSL version.
        const h2 = Math.max(0, Math.min(1, 0.5 + 0.5 * (a - b) / k));
        d = b * (1 - h2) + a * h2 + k * h2 * (1 - h2);
      }
    }
    return d;
  }

  // Camera math.
  cameraPosition(): [number, number, number] {
    const r = PLANET_RADIUS + this.player.altitude;
    const { lon, lat } = this.player;
    const cl = Math.cos(lat);
    return [Math.cos(lon) * cl * r, Math.sin(lat) * r, Math.sin(lon) * cl * r];
  }

  // Local "up" is the radial direction.
  private cameraBasis(): { right: number[], up: number[], fwd: number[] } {
    const pos = this.cameraPosition();
    const upR: [number, number, number] = [pos[0], pos[1], pos[2]];
    const ul = Math.hypot(upR[0], upR[1], upR[2]);
    const up = [upR[0] / ul, upR[1] / ul, upR[2] / ul];
    // Build a stable tangent basis. Use world Z as reference.
    let ref: [number, number, number] = [0, 0, 1];
    if (Math.abs(up[2]) > 0.95) ref = [1, 0, 0];
    // east = normalize(cross(up, ref))
    let ex = up[1] * ref[2] - up[2] * ref[1];
    let ey = up[2] * ref[0] - up[0] * ref[2];
    let ez = up[0] * ref[1] - up[1] * ref[0];
    const el = Math.hypot(ex, ey, ez); ex /= el; ey /= el; ez /= el;
    // north = cross(east, up)
    const nx = ey * up[2] - ez * up[1];
    const ny = ez * up[0] - ex * up[2];
    const nz = ex * up[1] - ey * up[0];
    // Apply yaw + pitch.
    const cy = Math.cos(this.player.yaw), sy = Math.sin(this.player.yaw);
    const cp = Math.cos(this.player.pitch), sp = Math.sin(this.player.pitch);
    // Heading vector in tangent plane: yaw rotates between north and east.
    const hx = nx * cy + ex * sy;
    const hy = ny * cy + ey * sy;
    const hz = nz * cy + ez * sy;
    // Forward tilts up/down from heading toward up.
    const fwd = [
      hx * cp + up[0] * sp,
      hy * cp + up[1] * sp,
      hz * cp + up[2] * sp,
    ];
    // right = normalize(cross(fwd, up))
    let rx = fwd[1] * up[2] - fwd[2] * up[1];
    let ry = fwd[2] * up[0] - fwd[0] * up[2];
    let rz = fwd[0] * up[1] - fwd[1] * up[0];
    const rl = Math.hypot(rx, ry, rz); rx /= rl; ry /= rl; rz /= rl;
    // recompute orthogonal "up" as cross(right, fwd) -> proper view up
    const ux = ry * fwd[2] - rz * fwd[1];
    const uy = rz * fwd[0] - rx * fwd[2];
    const uz = rx * fwd[1] - ry * fwd[0];
    return {
      right: [rx, ry, rz],
      up: [ux, uy, uz],
      fwd,
    };
  }

  cameraForward(): [number, number, number] {
    const b = this.cameraBasis();
    return [b.fwd[0], b.fwd[1], b.fwd[2]];
  }

  // Walk along the planet surface in the heading direction.
  move(forward: number, strafe: number, up: number, dt: number) {
    if (forward !== 0 || strafe !== 0) {
      // Convert linear motion into angular motion on the sphere.
      const speed = 2.5; // world units / second on surface
      const dist = speed * dt;
      // Compose movement vector in tangent plane.
      const b = this.cameraBasis();
      const pos = this.cameraPosition();
      const upR = [pos[0], pos[1], pos[2]];
      const ul = Math.hypot(upR[0], upR[1], upR[2]);
      const u = [upR[0] / ul, upR[1] / ul, upR[2] / ul];
      // Project fwd and right onto tangent plane.
      const projT = (v: number[]) => {
        const d = v[0] * u[0] + v[1] * u[1] + v[2] * u[2];
        return [v[0] - u[0] * d, v[1] - u[1] * d, v[2] - u[2] * d];
      };
      const ft = projT(b.fwd);
      const rt = projT(b.right);
      const ftl = Math.hypot(ft[0], ft[1], ft[2]) || 1;
      const rtl = Math.hypot(rt[0], rt[1], rt[2]) || 1;
      const mvx = (ft[0] / ftl) * forward + (rt[0] / rtl) * strafe;
      const mvy = (ft[1] / ftl) * forward + (rt[1] / rtl) * strafe;
      const mvz = (ft[2] / ftl) * forward + (rt[2] / rtl) * strafe;
      const ml = Math.hypot(mvx, mvy, mvz) || 1;
      const r = PLANET_RADIUS + this.player.altitude;
      const ang = dist / r;
      // New position = rotate current position toward (pos + mv*ang*r) along sphere.
      // Approximate by stepping in tangent then renormalizing.
      const npx = pos[0] + (mvx / ml) * ang * r;
      const npy = pos[1] + (mvy / ml) * ang * r;
      const npz = pos[2] + (mvz / ml) * ang * r;
      const nl = Math.hypot(npx, npy, npz);
      // Convert back to lon/lat at current altitude.
      const ny = npy / nl;
      this.player.lat = Math.asin(Math.max(-1, Math.min(1, ny)));
      this.player.lon = Math.atan2(npz / nl, npx / nl);
      this.markDirty();
    }
    if (up !== 0) {
      this.player.altitude = Math.max(0.5, Math.min(8, this.player.altitude + up * 3 * dt));
      this.markDirty();
    }
  }

  look(dx: number, dy: number) {
    // Standard FPS controls:
    //  - Mouse moves left (dx<0) → camera pans left → yaw decreases (using -=)
    //  - Mouse moves up   (dy<0) → camera tilts up   → pitch increases (using -=)
    this.player.yaw -= dx * 0.0025;
    this.player.pitch = Math.max(-1.4, Math.min(1.4, this.player.pitch - dy * 0.0025));
    this.markDirty();
  }

  rotateGhost(dx: number, dy: number) {
    // ghostUserQuat lives in CAMERA-LOCAL space, so the rotation axes are just
    // the canonical basis (X = camera right, Y = camera up). The ghost stays
    // visually stationary in the viewport as the player walks and looks
    // around — only E+mouse changes its appearance from the camera's POV.
    // We pre-multiply (q_new = q_delta * q_user) so the rotation feels like
    // grabbing the object in front of the camera.
    const qYaw   = quatFromAxisAngle(0, 1, 0, -dx * 0.01);
    const qPitch = quatFromAxisAngle(1, 0, 0, -dy * 0.01);
    this.ghostUserQuat = quatMul(quatMul(qYaw, qPitch), this.ghostUserQuat);
    const q = this.ghostUserQuat;
    const l = Math.hypot(q[0], q[1], q[2], q[3]) || 1;
    this.ghostUserQuat = [q[0]/l, q[1]/l, q[2]/l, q[3]/l];
    this.markDirty();
  }

  // Scroll wheel: push/pull ghost. Positive = farther.
  adjustGhostDistance(delta: number) {
    this.ghostDistance = Math.max(0.8, Math.min(15, this.ghostDistance + delta));
    this.markDirty();
  }

  setSubtractMode(on: boolean) {
    if (this.ghostSubtractMode !== on) {
      this.ghostSubtractMode = on;
      this.markDirty();
    }
  }

  setShape(k: ShapeKind) {
    this.selectedShape = k;
    // Sensible default sizes per shape.
    if (k === 0) this.shapeSize = [0.6, 0.6, 0.6];
    else if (k === 1) this.shapeSize = [0.5, 0.5, 0.5];
    else if (k === 2) this.shapeSize = [0.6, 0.18, 0.0];
    else this.shapeSize = [0.4, 0.6, 0.0];
    this.markDirty();
  }

  // Pack all (culled) edits + ghost into the GPU texture.
  private uploadEdits() {
    const cam = this.cameraPosition();
    // Evaluation culling: only consider edits whose AABB is near the camera /
    // ray reach. Anything beyond CULL_RADIUS is invisible from here anyway.
    let visible = cullEditsToSphere(this.edits, cam[0], cam[1], cam[2], CULL_RADIUS);
    // Hard cap for shader budget — keep most recent (player's current work).
    if (visible.length > MAX_EDITS_GPU - 1) {
      visible = visible.slice(visible.length - (MAX_EDITS_GPU - 1));
    }
    let idx = 0;
    const data = this.editsData;
    data.fill(0);
    for (let i = 0; i < visible.length; i++) {
      const e = visible[i];
      const o = idx * 16;
      // texel 0: pos.xyz, kind
      data[o + 0] = e.px; data[o + 1] = e.py; data[o + 2] = e.pz; data[o + 3] = e.kind;
      // texel 1: size.xyz, op
      data[o + 4] = e.sx; data[o + 5] = e.sy; data[o + 6] = e.sz; data[o + 7] = e.op;
      // texel 2: quat.xyzw
      data[o + 8] = e.qx; data[o + 9] = e.qy; data[o + 10] = e.qz; data[o + 11] = e.qw;
      // texel 3: (reserved=0, color.rgb) — read by shader as dd.yzw
      data[o + 12] = 0; data[o + 13] = e.cr; data[o + 14] = e.cg; data[o + 15] = e.cb;
      idx++;
    }
    const gl = this.gl;
    gl.bindTexture(gl.TEXTURE_2D, this.editsTex);
    gl.texSubImage2D(gl.TEXTURE_2D, 0, 0, 0, EDIT_TEX_WIDTH, EDIT_TEX_WIDTH, gl.RGBA, gl.FLOAT, this.editsData);
    return idx;
  }

  private resizeFbo(w: number, h: number) {
    if (w === this.fboW && h === this.fboH) return;
    const gl = this.gl;
    this.fboW = w; this.fboH = h;
    gl.bindTexture(gl.TEXTURE_2D, this.fboTex);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA8, w, h, 0, gl.RGBA, gl.UNSIGNED_BYTE, null);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
    gl.bindFramebuffer(gl.FRAMEBUFFER, this.fbo);
    gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0, gl.TEXTURE_2D, this.fboTex, 0);
    gl.bindFramebuffer(gl.FRAMEBUFFER, null);
  }

  // Render only when something changed (camera/edits/ghost). This is the single
  // biggest "potato" win — idle = 0 GPU work.
  renderIfDirty(timeMs: number) {
    if (!this.dirty) return false;
    this.dirty = false;
    this.render(timeMs);
    return true;
  }

  private render(timeMs: number) {
    const gl = this.gl;
    const canvas = gl.canvas as HTMLCanvasElement;
    const dpr = Math.min(window.devicePixelRatio || 1, 1.5);
    const cssW = canvas.clientWidth;
    const cssH = canvas.clientHeight;
    const fullW = Math.max(1, Math.floor(cssW * dpr));
    const fullH = Math.max(1, Math.floor(cssH * dpr));
    if (canvas.width !== fullW || canvas.height !== fullH) {
      canvas.width = fullW; canvas.height = fullH;
    }
    const lowW = Math.max(1, Math.floor(fullW * this.renderScale));
    const lowH = Math.max(1, Math.floor(fullH * this.renderScale));
    this.resizeFbo(lowW, lowH);

    // Pass 1: render at low res into FBO.
    const editCount = this.uploadEdits();
    gl.bindFramebuffer(gl.FRAMEBUFFER, this.fbo);
    gl.viewport(0, 0, lowW, lowH);
    gl.useProgram(this.prog);
    gl.bindVertexArray(this.vao);

    const cam = this.cameraPosition();
    const b = this.cameraBasis();
    gl.uniform2f(this.uniforms.uResolution!, lowW, lowH);
    gl.uniform3f(this.uniforms.uCamPos!, cam[0], cam[1], cam[2]);
    gl.uniformMatrix3fv(this.uniforms.uCamRot!, false, new Float32Array([
      b.right[0], b.right[1], b.right[2],
      b.up[0],    b.up[1],    b.up[2],
      -b.fwd[0], -b.fwd[1], -b.fwd[2],
    ]));
    gl.uniform1f(this.uniforms.uFovScale!, Math.tan(60 * Math.PI / 180 / 2));
    gl.uniform1f(this.uniforms.uPlanetRadius!, PLANET_RADIUS);
    gl.uniform1f(this.uniforms.uWorldRadius!, WORLD_RADIUS);
    gl.uniform1i(this.uniforms.uEditCount!, editCount);
    gl.uniform1i(this.uniforms.uEditTexWidth!, EDIT_TEX_WIDTH);
    gl.uniform1f(this.uniforms.uTime!, timeMs * 0.001);
    const sx = 0.5, sy = 0.7, sz = 0.5;
    const sl = Math.hypot(sx, sy, sz);
    gl.uniform3f(this.uniforms.uSunDir!, sx / sl, sy / sl, sz / sl);

    // Ghost uniforms (always visible — single shape, almost free vs. the scene loop).
    const gp = this.ghostWorldPos();
    const gq = this.ghostWorldQuat();
    gl.uniform1i(this.uniforms.uGhostKind!, this.selectedShape);
    gl.uniform3f(this.uniforms.uGhostPos!, gp[0], gp[1], gp[2]);
    gl.uniform3f(this.uniforms.uGhostSize!, this.shapeSize[0], this.shapeSize[1], this.shapeSize[2]);
    gl.uniform4f(this.uniforms.uGhostQuat!, gq[0], gq[1], gq[2], gq[3]);
    gl.uniform1f(this.uniforms.uGhostOp!, this.ghostSubtractMode ? -1 : 1);

    gl.activeTexture(gl.TEXTURE0);
    gl.bindTexture(gl.TEXTURE_2D, this.editsTex);
    gl.uniform1i(this.uniforms.uEdits!, 0);
    gl.drawArrays(gl.TRIANGLES, 0, 3);

    // Pass 2: blit upscaled to canvas.
    gl.bindFramebuffer(gl.FRAMEBUFFER, null);
    gl.viewport(0, 0, fullW, fullH);
    gl.useProgram(this.blitProg);
    gl.activeTexture(gl.TEXTURE0);
    gl.bindTexture(gl.TEXTURE_2D, this.fboTex);
    gl.uniform1i(this.blitUniformTex, 0);
    gl.drawArrays(gl.TRIANGLES, 0, 3);
  }

  // (No updateGhost() needed — ghost is always visible at a fixed distance
  // from the camera, recomputed inline in render() from current camera state.)

  setRenderScale(s: number) {
    this.renderScale = Math.max(0.25, Math.min(1.0, s));
    this.markDirty();
  }

  getStats() {
    return {
      totalEdits: this.edits.length,
      renderScale: this.renderScale,
      fboSize: [this.fboW, this.fboH] as [number, number],
    };
  }
}
